# Excel Export Format

## ExcelJS workbook template

```typescript
import ExcelJS from 'exceljs';

async function buildComplianceExport(data: ComplianceRow[], meta: ExportMeta) {
  const wb = new ExcelJS.Workbook();
  wb.creator = 'GovLink';
  wb.created = new Date();

  const ws = wb.addWorksheet('Compliance Scorecard');
  ws.views = [{ state: 'frozen', ySplit: 1 }];

  ws.columns = [
    { header: 'PSGC Code', key: 'psgc', width: 14 },
    { header: 'Barangay', key: 'barangay', width: 28 },
    { header: 'Requirement', key: 'requirement', width: 32 },
    { header: 'Due Date', key: 'due_date', width: 14 },
    { header: 'Status', key: 'status', width: 14 },
    { header: 'Pass/Fail', key: 'pass_fail', width: 12 },
    { header: 'Days Overdue', key: 'days_overdue', width: 14 },
  ];

  const headerRow = ws.getRow(1);
  headerRow.font = { bold: true };
  headerRow.fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: 'FFE2E8F0' } };
  headerRow.alignment = { vertical: 'middle' };
  ws.autoFilter = { from: 'A1', to: 'G1' };

  for (const row of data) {
    ws.addRow({
      ...row,
      pass_fail: row.pass_fail, // 'PASS' | 'FAIL' — explicit text
    });
  }

  addAuditLogSheet(wb, meta);
  addLegendSheet(wb);
  return wb;
}
```

## Pass/Fail column rules

| Condition | Pass/Fail value |
|-----------|-----------------|
| Accepted / on-time | `PASS` |
| Pending (not yet due) | `PENDING` |
| Overdue or returned | `FAIL` |

Use text values only — auditors print in grayscale. Optional conditional formatting is supplementary, not primary.

## Audit Log sheet

```typescript
function addAuditLogSheet(wb: ExcelJS.Workbook, meta: ExportMeta) {
  const log = wb.addWorksheet('Audit Log');
  const fields = [
    ['Report Type', meta.report_type],
    ['Generated At', formatManila(meta.generated_at)],
    ['Generated By', meta.generated_by_name],
    ['Role', meta.generated_by_role],
    ['Municipality', meta.municipality_name],
    ['Province', meta.province_name],
    ['Record Count', String(meta.record_count)],
    ['Export Document ID', meta.document_id],
    ['Content Hash', meta.content_hash],
    ['Verification URL', meta.verification_url],
    ['GovLink Version', meta.app_version],
  ];
  fields.forEach(([k, v], i) => {
    log.getCell(`A${i + 1}`).value = k;
    log.getCell(`A${i + 1}`).font = { bold: true };
    log.getCell(`B${i + 1}`).value = v;
  });
  log.getColumn(1).width = 22;
  log.getColumn(2).width = 48;
}
```

## Legend sheet

Define status codes, Pass/Fail criteria, and legal basis (RA 7160 section, DILG MC) for the report type.

## Filename convention

```
{report_type}_{municipality_slug}_{YYYYMMDD_HHmm}_PST.xlsx
```

Example: `compliance_scorecard_municipality-of-sample_20260822_1630_PST.xlsx`

## Download endpoint

```typescript
@UseGuards(JwtAuthGuard, RolesGuard)
@Roles(AppRole.MAYOR, AppRole.DEPT_HEAD)
@Get('exports/compliance-scorecard.xlsx')
async downloadCompliance(@TenantCtx() ctx: TenantContext, @Res() res: Response) {
  const buffer = await this.excelExport.buildComplianceScorecard(ctx);
  res.setHeader('Content-Type', 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet');
  res.setHeader('Content-Disposition', `attachment; filename="${filename}"`);
  res.send(buffer);
}
```
